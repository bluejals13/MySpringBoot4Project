# → Robert가 포함된 저자 검색
echo ""
echo "===== 4. 저자로 도서 검색 ====="
curl -s "http://localhost:8080/api/books/search/author?author=Robert"
echo ""

# 5. 제목으로 도서 검색
# GET /api/books/search/title?title=Clean
# → Clean이 포함된 제목 검색
echo ""
echo "===== 5. 제목으로 도서 검색 ====="
curl -s "http://localhost:8080/api/books/search/title?title=Clean"
echo ""

# 6. Book 부분 수정 - 제목
# PATCH /api/books/4
# → 제목만 변경하고 나머지 필드는 유지
echo ""
echo "===== 6. Book PATCH - 제목만 수정 ====="
curl -s -X PATCH http://localhost:8080/api/books/4 \
  -H "Content-Type: application/json" \
  -d '{"title":"Clean Code 수정본"}'
echo ""

# 7. Book 부분 수정 - 가격 + BookDetail 언어
# PATCH /api/books/4
# → price와 detailRequest.language만 변경
echo ""
echo "===== 7. Book PATCH - 가격 + 언어 수정 ====="
curl -s -X PATCH http://localhost:8080/api/books/4 \
  -H "Content-Type: application/json" \
  -d '{
    "price":15000,
    "detailRequest":{
      "language":"Korean"
    }
  }'
echo ""

# 8. BookDetail 부분 수정
# PATCH /api/books/4/detail
# → BookDetail의 description만 변경
echo ""
echo "===== 8. BookDetail PATCH - 설명만 수정 ====="
curl -s -X PATCH http://localhost:8080/api/books/4/detail \
  -H "Content-Type: application/json" \
  -d '{"description":"새로운 책 설명"}'
echo ""

# 9. 최종 조회
# → PATCH 결과가 실제로 반영되었는지 확인
echo ""
echo "===== 9. 최종 결과 확인 ====="
curl -s http://localhost:8080/api/books/4
echo ""
