echo "========================================"
echo " POST 테스트"
echo "========================================"

echo
echo "===== 11. Publisher 생성 ====="
curl -s -X POST http://localhost:8080/api/publishers \
  -H "Content-Type: application/json" \
  -d '{
    "name": "Manning Publications",
    "establishedDate": "1997-01-01",
    "address": "20 Baldwin Road, Shelter Island, NY"
  }'

echo

  -H "Content-Type: application/json" \

    "author": "Craig Walls",
    "isbn": "978-1617292545",
    "price": 45000,
    "publishDate": "2016-01-30",
